24    pub fn sample(&mut self, now: f64) -> f64 {
25        let t = if self.duration > 0.0 {
26            ((now - self.start) / self.duration).clamp(0.0, 1.0)

28            1.0

32        self.value = self.from + (target - self.from) * smooth;

34    }
